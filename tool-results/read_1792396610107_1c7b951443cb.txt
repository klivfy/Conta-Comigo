   253→function ccUtf8Bytes(str) {
   254→  try { return Array.from(new TextEncoder().encode(str)); }
   255→  catch (e) {
   256→    const bin = unescape(encodeURIComponent(str));
   257→    const out = [];
   258→    for (let i = 0; i < bin.length; i++) out.push(bin.charCodeAt(i) & 0xff);
   259→    return out;
   260→  }
   261→}
   262→function ccSha256Hex(input) {
   263→  const K = [0x428a2f98, 0x71374491, 0xb5c0fbcf, 0xe9b5dba5, 0x3956c25b, 0x59f111f1, 0x923f82a4, 0xab1c5ed5,

   269→    0x19a4c116, 0x1e376c08, 0x2748774c, 0x34b0bcb5, 0x391c0cb3, 0x4ed8aa4a, 0x5b9cca4f, 0x682e6ff3,
   270→    0x748f82ee, 0x78a5636f, 0x84c87814, 0x8cc70208, 0x90befffa, 0xa4506ceb, 0xbef9a3f7, 0xc67178f2];
   271→  const H0 = [0x6a09e667, 0xbb67ae85, 0x3c6ef372, 0xa54ff53a, 0x510e527f, 0x9b05688c, 0x1f83d9ab, 0x5be0cd19];
   272→  const rr = (v, a) => (((v >>> a) | (v << (32 - a))) >>> 0);
   273→  const bytes = ccUtf8Bytes(input);
   274→  const msg = bytes.slice();
   275→  msg.push(0x80);
   276→  while (msg.length % 64 !== 56) msg.push(0);
   277→  const bitsHi = Math.floor(bytes.length / 536870912);
   278→  const bitsLo = (bytes.length * 8) >>> 0;
   279→  msg.push((bitsHi >>> 24) & 255, (bitsHi >>> 16) & 255, (bitsHi >>> 8) & 255, bitsHi & 255,
   280→    (bitsLo >>> 24) & 255, (bitsLo >>> 16) & 255, (bitsLo >>> 8) & 255, bitsLo & 255);
   281→  const H = H0.slice();
   282→  const w = new Array(64);
   283→  for (let off = 0; off < msg.length; off += 64) {
   284→    for (let i = 0; i < 16; i++) {
   285→      const j = off + i * 4;
   286→      w[i] = ((msg[j] << 24) | (msg[j + 1] << 16) | (msg[j + 2] << 8) | msg[j + 3]) | 0;
   287→    }
   288→    for (let i = 16; i < 64; i++) {
   289→      const s0 = rr(w[i - 15], 7) ^ rr(w[i - 15], 18) ^ (w[i - 15] >>> 3);
   290→      const s1 = rr(w[i - 2], 17) ^ rr(w[i - 2], 19) ^ (w[i - 2] >>> 10);
   291→      w[i] = (w[i - 16] + s0 + w[i - 7] + s1) | 0;
   292→    }
   293→    let a = H[0], b = H[1], c = H[2], d = H[3], e = H[4], f = H[5], g = H[6], h = H[7];
   294→    for (let i = 0; i < 64; i++) {
   295→      const S1 = rr(e, 6) ^ rr(e, 11) ^ rr(e, 25);
   296→      const ch = (e & f) ^ (~e & g);
   297→      const t1 = (h + S1 + ch + K[i] + w[i]) | 0;
   298→      const S0 = rr(a, 2) ^ rr(a, 13) ^ rr(a, 22);
   299→      const maj = (a & b) ^ (a & c) ^ (b & c);
   300→      const t2 = (S0 + maj) | 0;
   301→      h = g; g = f; f = e; e = (d + t1) | 0;
   302→      d = c; c = b; b = a; a = (t1 + t2) | 0;
   303→    }
   304→    H[0] = (H[0] + a) | 0; H[1] = (H[1] + b) | 0; H[2] = (H[2] + c) | 0; H[3] = (H[3] + d) | 0;
   305→    H[4] = (H[4] + e) | 0; H[5] = (H[5] + f) | 0; H[6] = (H[6] + g) | 0; H[7] = (H[7] + h) | 0;
   306→  }
   307→  return H.map((x) => (x >>> 0).toString(16).padStart(8, '0')).join('');
   308→}

   351→  const cands = shuffle([answer + 1, answer - 1, answer + 2, answer - 2].concat(extras || []));
   352→  for (const c of cands) {
   353→    if (out.size >= 3) break;
   354→    if (c >= 0 && c !== answer && c <= cap && !out.has(c)) out.add(c);
   355→  }
   356→  let g = 3;
   357→  while (out.size < 3 && g <= 20) {
   358→    const up = answer + g, down = answer - g;
   359→    if (up <= cap && !out.has(up)) out.add(up);
   360→    else if (down >= 0 && !out.has(down)) out.add(down);
   361→    g++;
   362→  }
   363→  return shuffle([answer].concat([...out].slice(0, 3)));
   364→}
