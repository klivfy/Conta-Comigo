   141→function ensureMissions() {
   142→  const p = P(); const today = fmtDate();
   143→  if (!p.today || p.today.d !== today) p.today = { d: today, rounds: 0, done: false, modes: {} };
   144→  if (p.missions.d !== today) {
   145→    let seed = 0; for (const ch of today) seed += ch.charCodeAt(0);
   146→    const idx = shuffle([0, 1, 2, 3, 4].map((x) => x)).sort((a, b) => ((seed * (a + 3)) % 7) - ((seed * (b + 3)) % 7)).slice(0, 3);
   147→    p.missions = { d: today, list: idx.map((i) => ({ id: MISSION_DEFS[i].id, p: 0, done: false })) };
   148→    saveState();
   149→  }
   150→}
   151→function missionEvent(ev, n) {
   152→  const p = P(); const defs = { rounds2: 'round', first5: 'first', combo3: 'combo', newmode: 'newmode', review3: 'review' };
   153→  let changed = false;
   154→  p.missions.list.forEach((m) => {
   155→    const def = MISSION_DEFS.find((d) => d.id === m.id);
   156→    if (!def || m.done || defs[m.id] !== ev) return;
   157→    m.p = Math.min(def.goal, m.p + (n || 1));
   158→    if (m.p >= def.goal) {
   159→      m.done = true; changed = true;
   160→      p.stars += 3; p.missionsDone++;
   161→      toast('🎉 ' + t('mission_done'));
   162→      SFX.star(); refreshChrome();
   163→    }
   164→  });
   165→  if (changed) saveState();
   166→}

   379→/* ---------- cronômetro de pontos (v2.2) ----------
   380→   Barra suave por questão: pausa em overlays/aba oculta/tela inativa,
   381→   NUNCA pune — ao esgotar só zera o bônus de rapidez. Avaliação (diag)
   382→   e duelo ficam sem cronômetro (menos pressão). */
   383→const Timer = {
   384→  iv: null, totalMs: 0, leftMs: 0, last: 0, active: false,
   385→  allowed() { return !!G().timer && R.kind !== 'diag' && R.kind !== 'duel'; },
   386→  start(sec) {
   387→    this.stop();
   388→    const wrap = $('#timer-wrap');
   389→    if (!wrap) return;
   390→    if (!this.allowed()) { wrap.classList.add('hidden'); this.totalMs = 0; this.leftMs = 0; return; }
   391→    this.totalMs = sec * 1000; this.leftMs = this.totalMs; this.last = Date.now(); this.active = true;
   392→    wrap.classList.remove('hidden', 'warn', 'low', 'out');
   393→    wrap.setAttribute('aria-label', t('s_timer'));
   394→    this.paint();
   395→    this.iv = setInterval(() => this.tick(), 200);
   396→  },
   397→  paused() {
   398→    const scr = $('#screen-game');
   399→    return !scr || !scr.classList.contains('active') || overlayOpen() || document.hidden ||
   400→      $('#answers').classList.contains('locked');
   401→  },
   402→  tick() {
   403→    if (!this.active) return;
   404→    const now = Date.now(); const dt = now - this.last; this.last = now;
   405→    if (!this.paused()) this.leftMs = Math.max(0, this.leftMs - dt);
   406→    this.paint();
   407→    if (this.leftMs <= 0) {
   408→      this.active = false; clearInterval(this.iv); this.iv = null;
   409→      const w = $('#timer-wrap'); if (w) w.classList.add('out');
   410→      if (!R.locked && R.current) showBubble('⏰', t('time_over'));
   411→    }
   412→  },
   413→  paint() {
   414→    const w = $('#timer-wrap'); if (!w) return;
   415→    const pct = this.totalMs ? Math.round(100 * this.leftMs / this.totalMs) : 0;
   416→    const bar = $('#timer-bar');
   417→    if (bar) bar.style.width = pct + '%';
   418→    w.setAttribute('aria-valuenow', String(pct));
   419→    w.classList.toggle('warn', pct < 50 && pct >= 25);
   420→    w.classList.toggle('low', pct < 25 && pct > 0);
   421→  },
   422→  stop() { this.active = false; if (this.iv) { clearInterval(this.iv); this.iv = null; } },
   423→  hide() { this.stop(); const w = $('#timer-wrap'); if (w) w.classList.add('hidden'); },
   424→  ratio() { return this.totalMs ? clamp(this.leftMs / this.totalMs, 0, 1) : 0; },

   475→  ensureAudio();

   477→  const v = btn.dataset.v;

   479→  const mode = currentMode();

   481→
   482→  if (correct) {
   483→    R.locked = true;
   484→    const speedRatio = Timer.ratio();
   485→    const p = P();
   486→    let gained = firstTry ? 2 : 1;
   487→    if (R.kind === 'boss') gained *= 2;
   488→    if (firstTry) {
   489→      R.firsts++;
   490→      p.firsts++;
   491→      R.combo++;
   492→      if (R.kind === 'diag') R.diagOK++;
   493→      p.fatigue = 0;

   497→      missionEvent('first');
   498→    }
   499→    grantStars(gained);
   500→    p.rights++;
   501→    bumpLvlStat(mode, R.level, true);
   502→    if (mode === 'dividir') p.firstDiv = true;
   503→    if (R.kind === 'review') { gradeReview(R.due[R.qIndex], true); missionEvent('review'); }
   504→
   505→    // v2.2: pontos por rapidez + precisão (fora de diagnóstico e duelo)
   506→    if (R.kind !== 'diag' && R.kind !== 'duel') {
   507→      const mult = clamp(R.level || 1, 1, MAX_LEVEL);
   508→      let pts = firstTry ? Math.round(100 * mult + 60 * mult * speedRatio) : 40 * mult;
   509→      if (R.kind === 'boss') pts *= 2;
   510→      grantPts(pts);
   511→    }
   512→    Timer.stop();
   513→
   514→    btn.classList.add('good');

   521→
   522→    // S07: adaptação por habilidade (fora de diagnóstico/duelo)
   523→    if (R.kind === 'normal' || R.kind === 'boss') {
   524→      const grew = recordResult(mode, firstTry);
   525→      if (grew > 0) setTimeout(() => levelUpOverlay(mode, p.levels[mode]), 1500);
   526→    }

   528→
   529→    R.history.push({ txt: R.current.display, ok: firstTry });
   530→    R.qIndex++;
   531→    setTimeout(() => {
   532→      if (R.qIndex >= R.total) endRound();
   533→      else { if (R.kind === 'duel') R.duel.turn = R.duel.turn === 1 ? 2 : 1; nextQuestion(); }
   534→    }, 1600);
   535→  } else {
   536→    R.misses++;
   537→    P().wrongs++;
   538→    P().fatigue++;
   539→    bumpLvlStat(mode, R.level, false);
   540→    if ((R.kind === 'normal' || R.kind === 'duel') && R.current.f) {
   541→      addReviewCard(mode === 'mistar' ? R.current.underlying || mode : mode, R.level, R.current.f);
   542→    }
   543→    SFX.wrong();
   544→    btn.classList.add('bad'); btn.disabled = true;

   548→        if (good) good.classList.add('hint');
   549→        showBubble('👉', t('hint'));
   550→        TTS.speak(t('hint'));

   552→        const r = pick(I18N[LANG].retry);
   553→        showBubble('🙈', r); TTS.speak(r);
   554→      }
   555→    }

   568→function endRound() {
   569→  const p = P();
   570→  const today = fmtDate();
   571→  // ofensiva S15
   572→  if (p.lastPlay !== today) {
   573→    p.streak = (p.lastPlay === yesterdayStr()) ? (p.streak || 0) + 1 : 1;
   574→    p.lastPlay = today;
   575→  }
   576→  // semanas (S26)
   577→  const wk = weekKey();
   578→  let wrow = p.weeks.find((w) => w.k === wk);
   579→  if (!wrow) { wrow = { k: wk, rounds: 0, rights: 0 }; p.weeks.push(wrow); p.weeks = p.weeks.slice(-8); }
   580→  wrow.rounds++; wrow.rights += R.firsts;
   581→  // rodadas e meta S11
   582→  p.rounds++;
   583→  if (!p.today) ensureMissions();
   584→  p.today.rounds++;
   585→  missionEvent('round');
   586→  if (G().goalRounds > 0 && p.today.rounds >= G().goalRounds && !p.today.done) {
   587→    p.today.done = true; grantStars(5); setTimeout(() => toast('🎯 ' + t('goal_done')), 800);
   588→  }
   589→  // chefe S18

   594→  } else if (R.kind === 'normal' && p.rounds % 5 === 0) {

   821→  try {

   824→  } catch (e) { /* ignora */ }

   888→function onKey(e) {
   889→  if (overlayOpen()) {
   890→    if (e.key === 'Enter') { const b = $('#overlay-card .big-btn:not([disabled])'); if (b) b.click(); }
   891→    return;
   892→  }
   893→  if ($('#screen-game').classList.contains('active')) {
   894→    if (e.key >= '1' && e.key <= '4') { const b = $$('#answers .ans')[Number(e.key) - 1]; if (b) b.click(); }
   895→    else if (e.key === 'Enter' || e.key.toLowerCase() === 'r') { if (R.current) TTS.speak(R.current.tts); }
   896→    else if (e.key.toLowerCase() === 'h') $('#btn-home').click();
   897→  } else if (e.key.toLowerCase() === 'm') $('#btn-mute').click();
   898→  if (e.key === 'Escape') closePopovers();
   899→}
