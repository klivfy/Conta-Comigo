   123→      if (ds <= p.lastPlay) {
   124→        const diffDays = Math.round((end - d) / 86400000);
   125→        played = diffDays >= 0 && diffDays < p.streak;
   126→      }
   127→    }

   141→function ensureMissions() {
   142→  const p = P(); const today = fmtDate();
   143→  if (!p.today || p.today.d !== today) p.today = { d: today, rounds: 0, done: false, modes: {} };
   144→  if (p.missions.d !== today) {
   145→    let seed = 0; for (const ch of today) seed += ch.charCodeAt(0);
   146→    const idx = shuffle([0, 1, 2, 3, 4].map((x) => x)).sort((a, b) => ((seed * (a + 3)) % 7) - ((seed * (b + 3)) % 7)).slice(0, 3);
   147→    p.missions = { d: today, list: idx.map((i) => ({ id: MISSION_DEFS[i].id, p: 0, done: false })) };
   148→    saveState();
   149→  }
   150→}
   151→function missionEvent(ev, n) {
   152→  const p = P(); const defs = { rounds2: 'round', first5: 'first', combo3: 'combo', newmode: 'newmode', review3: 'review' };
   153→  let changed = false;
   154→  p.missions.list.forEach((m) => {
   155→    const def = MISSION_DEFS.find((d) => d.id === m.id);
   156→    if (!def || m.done || defs[m.id] !== ev) return;
   157→    m.p = Math.min(def.goal, m.p + (n || 1));
   158→    if (m.p >= def.goal) {
   159→      m.done = true; changed = true;
   160→      p.stars += 3; p.missionsDone++;
   161→      toast('🎉 ' + t('mission_done'));
   162→      SFX.star(); refreshChrome();
   163→    }
   164→  });
   165→  if (changed) saveState();
   166→}

   396→function onAnswer(btn) {
   397→  ensureAudio();
   398→  if (R.locked || !R.current) return;
   399→  const v = btn.dataset.v;
   400→  const correct = R.current.kind === 'cmp' ? v === String(R.current.answer) : Number(v) === R.current.answer;
   401→  const mode = currentMode();
   402→  const firstTry = R.misses === 0;
   403→
   404→  if (correct) {
   405→    R.locked = true;
   406→    const p = P();
   407→    let gained = firstTry ? 2 : 1;
   408→    if (R.kind === 'boss') gained *= 2;
   409→    if (firstTry) {
   410→      R.firsts++;
   411→      p.firsts++;
   412→      R.combo++;
   413→      if (R.kind === 'diag') R.diagOK++;
   414→      p.fatigue = 0;
   415→      if (R.combo > (p.bestCombo || 0)) p.bestCombo = R.combo;
   416→      if (R.combo === 3) missionEvent('combo');
   417→      if (R.combo > 0 && R.combo % 5 === 0) { grantStars(2); toast('🔥 +2 ⭐'); }
   418→      missionEvent('first');
   419→    }
   420→    grantStars(gained);
   421→    p.rights++;
   422→    bumpLvlStat(mode, R.level, true);
   423→    if (mode === 'dividir') p.firstDiv = true;
   424→    if (R.kind === 'review') { gradeReview(R.due[R.qIndex], true); missionEvent('review'); }
   425→
   426→    btn.classList.add('good');
   427→    $('#answers').classList.add('locked');
   428→    SFX.correct(); if (firstTry) SFX.star();
   429→    confetti(firstTry ? 30 : 18);
   430→    const pr = pick(I18N[LANG].praise);
   431→    showBubble(mascotStage().emoji, pr);
   432→    TTS.speak(pr);
   433→
   434→    // S07: adaptação por habilidade (fora de diagnóstico/duelo)
   435→    if (R.kind === 'normal' || R.kind === 'boss') {
   436→      const grew = recordResult(mode, firstTry);

   480→function endRound() {
   481→  const p = P();
   482→  const today = fmtDate();
   483→  // ofensiva S15
   484→  if (p.lastPlay !== today) {
   485→    p.streak = (p.lastPlay === yesterdayStr()) ? (p.streak || 0) + 1 : 1;
   486→    p.lastPlay = today;
   487→  }
   488→  // semanas (S26)
   489→  const wk = weekKey();
   490→  let wrow = p.weeks.find((w) => w.k === wk);
   491→  if (!wrow) { wrow = { k: wk, rounds: 0, rights: 0 }; p.weeks.push(wrow); p.weeks = p.weeks.slice(-8); }
   492→  wrow.rounds++; wrow.rights += R.firsts;
   493→  // rodadas e meta S11
   494→  p.rounds++;
   495→  if (!p.today) ensureMissions();
   496→  p.today.rounds++;
   497→  missionEvent('round');

   506→  } else if (R.kind === 'normal' && p.rounds % 5 === 0) {

   509→  checkAch();
   510→  saveState(true);
   511→
   512→  show('end');
   513→  SFX.fanfare();

   515→  $('#end-title').textContent = R.stars >= 8 ? t('end_great') : t('end_good');
