   138→function ensureMissions() {
   139→  const p = P(); const today = fmtDate();
   140→  if (!p.today || p.today.d !== today) p.today = { d: today, rounds: 0, done: false, modes: {} };
   141→  if (p.missions.d !== today) {
   142→    let seed = 0; for (const ch of today) seed += ch.charCodeAt(0);
   143→    const idx = shuffle([0, 1, 2, 3, 4].map((x) => x)).sort((a, b) => ((seed * (a + 3)) % 7) - ((seed * (b + 3)) % 7)).slice(0, 3);
   144→    p.missions = { d: today, list: idx.map((i) => ({ id: MISSION_DEFS[i].id, p: 0, done: false })) };
   145→    saveState();
   146→  }
   147→}
   148→function missionEvent(ev, n) {
   149→  const p = P(); const defs = { rounds2: 'round', first5: 'first', combo3: 'combo', newmode: 'newmode', review3: 'review' };
   150→  let changed = false;
   151→  p.missions.list.forEach((m) => {
   152→    const def = MISSION_DEFS.find((d) => d.id === m.id);
   153→    if (!def || m.done || defs[m.id] !== ev) return;
   154→    m.p = Math.min(def.goal, m.p + (n || 1));
   155→    if (m.p >= def.goal) {
   156→      m.done = true; changed = true;
   157→      p.stars += 3; p.missionsDone++;
   158→      toast('🎉 ' + t('mission_done'));
   159→      SFX.star(); refreshChrome();
   160→    }
   161→  });
   162→  if (changed) saveState();
   163→}

   393→function onAnswer(btn) {
   394→  ensureAudio();
   395→  if (R.locked || !R.current) return;
   396→  const v = btn.dataset.v;
   397→  const correct = R.current.kind === 'cmp' ? v === String(R.current.answer) : Number(v) === R.current.answer;
   398→  const mode = currentMode();
   399→  const firstTry = R.misses === 0;
   400→
   401→  if (correct) {
   402→    R.locked = true;
   403→    const p = P();
   404→    let gained = firstTry ? 2 : 1;
   405→    if (R.kind === 'boss') gained *= 2;
   406→    if (firstTry) {
   407→      R.firsts++;
   408→      p.firsts++;
   409→      R.combo++;
   410→      if (R.kind === 'diag') R.diagOK++;
   411→      p.fatigue = 0;
   412→      if (R.combo > (p.bestCombo || 0)) p.bestCombo = R.combo;
   413→      if (R.combo === 3) missionEvent('combo');
   414→      if (R.combo > 0 && R.combo % 5 === 0) { grantStars(2); toast('🔥 +2 ⭐'); }
   415→      missionEvent('first');
   416→    }
   417→    grantStars(gained);
   418→    p.rights++;
   419→    bumpLvlStat(mode, R.level, true);
   420→    if (mode === 'dividir') p.firstDiv = true;
   421→    if (R.kind === 'review') { gradeReview(R.due[R.qIndex], true); missionEvent('review'); }
   422→
   423→    btn.classList.add('good');
   424→    $('#answers').classList.add('locked');
   425→    SFX.correct(); if (firstTry) SFX.star();
   426→    confetti(firstTry ? 30 : 18);
   427→    const pr = pick(I18N[LANG].praise);
   428→    showBubble(mascotStage().emoji, pr);
   429→    TTS.speak(pr);
   430→
   431→    // S07: adaptação por habilidade (fora de diagnóstico/duelo)
   432→    if (R.kind === 'normal' || R.kind === 'boss') {
   433→      const grew = recordResult(mode, firstTry);
   434→      if (grew > 0) setTimeout(() => levelUpOverlay(mode, p.levels[mode]), 1500);
   435→    }

   477→function endRound() {
   478→  const p = P();
   479→  const today = fmtDate();
   480→  // ofensiva S15
   481→  if (p.lastPlay !== today) {
   482→    p.streak = (p.lastPlay === yesterdayStr()) ? (p.streak || 0) + 1 : 1;
   483→    p.lastPlay = today;
   484→  }
   485→  // semanas (S26)
   486→  const wk = weekKey();
   487→  let wrow = p.weeks.find((w) => w.k === wk);
   488→  if (!wrow) { wrow = { k: wk, rounds: 0, rights: 0 }; p.weeks.push(wrow); p.weeks = p.weeks.slice(-8); }
   489→  wrow.rounds++; wrow.rights += R.firsts;
   490→  // rodadas e meta S11

   509→  show('end');
