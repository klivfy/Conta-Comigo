   141→   141→function ensureMissions() {
   142→   142→  const p = P(); const today = fmtDate();
   143→   143→  if (!p.today || p.today.d !== today) p.today = { d: today, rounds: 0, done: false, modes: {} };
   144→   144→  if (p.missions.d !== today) {
   145→   145→    let seed = 0; for (const ch of today) seed += ch.charCodeAt(0);
   146→   146→    const idx = shuffle([0, 1, 2, 3, 4].map((x) => x)).sort((a, b) => ((seed * (a + 3)) % 7) - ((seed * (b + 3)) % 7)).slice(0, 3);
   147→   147→    p.missions = { d: today, list: idx.map((i) => ({ id: MISSION_DEFS[i].id, p: 0, done: false })) };
   148→   148→    saveState();
   149→   149→  }

   423→   423→  hide() { this.stop(); const w = $('#timer-wrap'); if (w) w.classList.add('hidden'); },
